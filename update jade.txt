/* ---------- Password hashing: salted PBKDF2 ---------- */
99
const b64=a=>btoa(String.fromCharCode(...a));

const unb64=s=>Uint8Array.from(atob(s),c=>c.charCodeAt(0));

const newSalt=()=>b64(crypto.getRandomValues(new Uint8Array(16)));

async function hash(pw,salt){
103
  if(window.crypto&&crypto.subtle){
104
    const k=await crypto.subtle.importKey('raw',new TextEncoder().encode(pw),'PBKDF2',false,['deriveBits']);
105
    const bits=await crypto.subtle.deriveBits({name:'PBKDF2',salt:unb64(salt),iterations:100000,hash:'SHA-256'},k,256);
106
    return b64(new Uint8Array(bits));
107
  }
108
  let h=0;for(const c of salt+pw)h=Math.imul(31,h)+c.charCodeAt(0)|0;return String(h); // fallback only
109
}

function inPeriod(dt,p){
139
  const t=today();
140
  if(p==='yearly')return dt.slice(0,4)===t.slice(0,4);
141
  if(p==='monthly')return dt.slice(0,7)===t.slice(0,7);
142
  const diff=(new Date(t)-new Date(dt))/864e5;return diff>=0&&diff<7;
143
}


# VALORANT TOURNAMENT MANAGEMENT SYSTEM

teams = []

while True:
    print("\n==============================")
    print("   VALORANT TOURNAMENT SYSTEM")
    print("==============================")
    print("1. Register Team")
    print("2. View Registered Teams")
    print("3. Start Tournament")
    print("4. Exit")

    choice = input("\nEnter your choice: ")

    if choice == "1":
        team_name = input("Enter Team Name: ")
        captain = input("Enter Team Captain: ")

        teams.append({
            "name": team_name,
            "captain": captain
        })

        print("Team successfully registered!")

    elif choice == "2":
        print("\n=== REGISTERED TEAMS ===")

        if len(teams) == 0:
            print("No teams registered yet.")
        else:
            for i, team in enumerate(teams, start=1):
                print(f"{i}. {team['name']} - Captain: {team['captain']}")

    elif choice == "3":
        print("\n=== TOURNAMENT PARTICIPANTS ===")

        if len(teams) < 2:
            print("You need at least 2 teams to start!")
        else:
            print("Tournament is ready!")
            for team in teams:
                print("-", team["name"])

    elif choice == "4":
        print("Thank you for using VALORANT Tournament System!")
        break

    else:
        print("Invalid choice. Please try again.")
